# Version resource compiled into ContactsManager.exe.
#
# Without this, Windows shows the file as having no name and no
# description: the Properties page is blank, the taskbar has nothing to
# call it, and SmartScreen's "More info" screen has nothing to report.
# Filling it in does not remove the SmartScreen warning — only a signing
# certificate does that — but it does mean the file says plainly what it
# is when anyone stops to look. The copyright and licence go here
# too, so they are visible in Properties without running anything.
#
# Built by PyInstaller from ContactsManager.spec.

VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          '040904B0',
          [
            StringStruct('FileDescription',
                         'Contacts Manager - manage contacts offline, '
                         'with CSV and vCard import and export'),
            StringStruct('FileVersion', '1.0.0.0'),
            StringStruct('InternalName', 'ContactsManager'),
            StringStruct('OriginalFilename', 'ContactsManager.exe'),
            StringStruct('ProductName', 'Contacts Manager'),
            StringStruct('ProductVersion', '1.0.0.0'),
            StringStruct('LegalCopyright',
                         'Copyright (C) 2026 Mahmoud Shrawy'),
            StringStruct('Comments',
                         'Portable. Keeps its data in a folder beside '
                         'itself. Built for screen reader users; works '
                         'in English and Arabic. Free software under the '
                         'GNU General Public License, version 3 or later; '
                         'see the LICENSE file.'),
          ]
        )
      ]
    ),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
